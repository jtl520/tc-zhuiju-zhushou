    function textOf(selector) {
        const node = document.querySelector(selector);
        return cleanText(node?.getAttribute('title') || node?.textContent || '');
    }

    function firstText(selectors) {
        for (const selector of selectors || []) {
            const text = textOf(selector);
            if (text) return text;
        }
        return '';
    }

    function findVisibleTitleCandidate() {
        const candidates = [...document.querySelectorAll(
            'h1, h2, h3, h4, [data-title], [class*="title" i], [class*="name" i]'
        )];
        let best = {score: 0, text: ''};

        candidates.forEach((node) => {
            const text = cleanText(node.getAttribute('data-title') || node.textContent);
            if (!text || text.length < 2 || text.length > 100 || node.offsetParent === null) return;
            if (isGenericMediaTitle(text)) return;

            const className = String(node.className || '');
            const rect = node.getBoundingClientRect();
            let score = 1;
            if (/第\s*\d+\s*[集期话]/.test(text)) score += 40;
            if (/player|play|video|album|detail|intro/i.test(className)) score += 20;
            if (rect.top >= 0 && rect.top < 220) score += 12;
            if (rect.left > window.innerWidth * 0.58 && rect.top >= 0 && rect.top < 360) score += 15;
            if (text.length >= 2 && text.length <= 20) score += 8;
            if (text.length >= 4 && text.length <= 60) score += 5;
            if (score > best.score) best = {score, text};
        });

        return best.text;
    }

    function cleanText(value) {
        return String(value || '')
            .replace(/\s+/g, ' ')
            .replace(/^\s+|\s+$/g, '')
            .slice(0, MAX_TITLE_LENGTH);
    }

    function isGenericMediaTitle(value) {
        const text = cleanText(value);
        if (!text) return true;
        return /^(?:腾讯视频|爱奇艺|优酷|芒果TV|哔哩哔哩|bilibili|首页|电视剧|电影|综艺|动漫|视频|讨论|选集|播放列表|相关推荐|周边视频)$/i.test(text)
            || /^(?:正片\s*)?(?:第\s*)?\d{1,4}\s*[集期话](?:上|中|下|完结|纯享版|加更版)?(?:\s|$|[：:—-])/i.test(text)
            || /^(?:爱奇艺|腾讯视频)\s*[-—_：:]?\s*(?:在线视频网站|海量正版|热门视频).*$/i.test(text)
            || /^(?:爱奇艺|腾讯视频).*(?:海量正版|高清在线观看).*$/i.test(text);
    }

    function readNestedValue(root, path) {
        return path.split('.').reduce((value, key) => value?.[key], root);
    }

    function findStructuredString(root, preferredKeys) {
        if (!root || typeof root !== 'object') return '';
        const queue = [{value: root, depth: 0}];
        const visited = new WeakSet();
        let checked = 0;

        while (queue.length && checked < 12000) {
            const {value, depth} = queue.shift();
            if (!value || typeof value !== 'object' || visited.has(value)) continue;
            visited.add(value);
            checked += 1;

            for (const key of preferredKeys) {
                const candidate = value[key];
                if (
                    typeof candidate === 'string'
                    && candidate.length >= 2
                    && candidate.length <= 100
                    && !isGenericMediaTitle(candidate)
                ) {
                    return cleanText(candidate);
                }
            }
            if (depth >= 8) continue;

            Object.entries(value).forEach(([key, child]) => {
                if (/recommend|related|suggest|rank|hot/i.test(key)) return;
                if (child && typeof child === 'object') {
                    queue.push({value: child, depth: depth + 1});
                }
            });
        }
        return '';
    }

    function findStructuredStringInScripts(preferredKeys) {
        const keyPattern = preferredKeys
            .map((key) => key.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
            .join('|');
        if (!keyPattern) return '';
        const pattern = new RegExp(`"(?:${keyPattern})"\\s*:\\s*"((?:\\\\.|[^"\\\\])*)"`, 'i');

        for (const script of document.scripts) {
            const source = script.textContent || '';
            if (!source || source.length > 6_000_000) continue;
            const match = source.match(pattern);
            if (!match) continue;
            try {
                const candidate = JSON.parse(`"${match[1]}"`);
                if (!isGenericMediaTitle(candidate)) return cleanText(candidate);
            } catch (error) {
                // 忽略无法解码的内嵌数据，继续检查其他脚本。
            }
        }
        return '';
    }

    function extractStructuredPlatformTitle(platform) {
        const page = typeof unsafeWindow === 'object' ? unsafeWindow : window;
        if (platform?.name === '哔哩哔哩') {
            const directPaths = [
                '__INITIAL_STATE__.mediaInfo.title',
                '__INITIAL_STATE__.mediaInfo.season_title',
                '__INITIAL_STATE__.epInfo.season_title',
                '__INITIAL_STATE__.seasonInfo.title'
            ];
            for (const path of directPaths) {
                const candidate = readNestedValue(page, path);
                if (typeof candidate === 'string' && !isGenericMediaTitle(candidate)) {
                    return cleanText(candidate);
                }
            }
            const keys = ['season_title', 'seasonTitle', 'media_title', 'mediaTitle'];
            return findStructuredString(page.__INITIAL_STATE__, keys)
                || findStructuredStringInScripts(keys);
        }

        if (platform?.name === '优酷') {
            const roots = [
                page.__INITIAL_DATA__,
                page.__INITIAL_STATE__,
                page.__NEXT_DATA__,
                page.__NUXT__
            ];
            const keys = [
                'showName',
                'show_name',
                'seriesName',
                'series_name',
                'albumName',
                'album_name',
                'programName',
                'showTitle'
            ];
            for (const root of roots) {
                const candidate = findStructuredString(root, keys);
                if (candidate) return candidate;
            }
            return findStructuredStringInScripts(keys);
        }

        if (platform?.name === '腾讯视频') {
            const roots = [
                page.__INITIAL_DATA__,
                page.__INITIAL_STATE__,
                page.__NEXT_DATA__,
                page.__PINIA__,
                page.__NUXT__,
                page.__PLAYER_CONFIG__
            ];
            const keys = [
                'coverTitle',
                'cover_title',
                'albumName',
                'album_name',
                'showName',
                'show_name',
                'seriesName',
                'series_name',
                'programName',
                'columnName'
            ];
            for (const root of roots) {
                const candidate = findStructuredString(root, keys);
                if (candidate) return candidate;
            }
            return findStructuredStringInScripts(keys);
        }
        return '';
    }

    let structuredTitleCache = {url: '', title: '', checkedAt: 0};

    function getStructuredPlatformTitle(platform) {
        if (
            structuredTitleCache.url === location.href
            && (structuredTitleCache.title || Date.now() - structuredTitleCache.checkedAt < 3000)
        ) {
            return structuredTitleCache.title;
        }
        structuredTitleCache = {
            url: location.href,
            title: extractStructuredPlatformTitle(platform),
            checkedAt: Date.now()
        };
        return structuredTitleCache.title;
    }

    function metaContent(...selectors) {
        for (const selector of selectors) {

            if (value) return cleanText(value);
        }
        return '';
    }

    function stripTitleDecorations(value) {
        return cleanText(value)
            .replace(/^[《〈「『【〔“"'‘]+\s*/, '')
            .replace(/\s*[》〉」』】〕”"'’]+$/, '');
    }

    function normalizeDocumentTitle(value) {
        return stripTitleDecorations(
            cleanText(value)
            .replace(/^(.+?)_0*\d{1,4}_(?:电视剧|电影|综艺|动漫).*$/i, '$1')
            .replace(/\s*[-_|—]\s*(腾讯视频|爱奇艺|优酷|芒果TV|哔哩哔哩|bilibili).*$/i, '')
            .replace(/\s*在线观看.*$/i, '')
            .replace(/\s*高清视频.*$/i, '')
            .replace(/(?:简介|详情)$/, '')
        );
    }

    function chooseBestMediaTitle(candidates) {
        const normalized = candidates
            .map((value) => normalizeDocumentTitle(value))
            .filter((title) => title && !isGenericMediaTitle(title));
        const reliable = normalized.find((title) => (
            !/^第\s*\d{1,2}\s*季$/.test(title)
            && !/^(?:正片\s*)?第\s*\d{1,4}\s*[集期话]/.test(title)
            && !/^(?:特别加更|加更|纯享|先导|预告|花絮|彩蛋)(?:\s|[：:]|$)/.test(title)
        ));
        return reliable || normalized[0] || '';
    }

    function extractPlatformDocumentTitle(platform) {
        const tabTitle = cleanText(document.title);
        if (!tabTitle || isGenericMediaTitle(tabTitle)) return '';

        const numberedTitle = tabTitle.match(
            /^(.+?)(?:[_\s-]+)(?:第)?0*(\d{1,4})(?:\s*[集期话])?(?:[_\s-]+)(?:电视剧|电影|综艺|动漫|国创|番剧)/
        );
        if (numberedTitle) return cleanText(numberedTitle[1]);

        const intent = parseSearchIntent(tabTitle);
        if (intent.episodeNumber && intent.title && !isGenericMediaTitle(intent.title)) {
            return intent.title;
        }

        if (['哔哩哔哩', '优酷', '腾讯视频'].includes(platform?.name)) {
            const baseTitle = cleanText(
                tabTitle.split(/\s*[-_|—]\s*(?:电视剧|电影|综艺|动漫|国创|番剧|高清|优酷|腾讯视频|bilibili|哔哩哔哩)/i)[0]
            );
            return isGenericMediaTitle(baseTitle) ? '' : baseTitle;
        }
        return '';
    }

    function parseSearchIntent(value) {
        const query = cleanText(value);
        const episodeMatch = query.match(
            /第\s*(\d{1,4})\s*([集期话部])|(?:^|\s)EP?\s*0*(\d{1,4})(?:\s|$)/i
        );
        const episodeNumber = episodeMatch
            ? Number(episodeMatch[1] || episodeMatch[3])
            : null;
        const indexUnit = episodeMatch
            ? (episodeMatch[2] || '集')
            : '';
        const title = cleanText(
            episodeMatch
                ? query.slice(0, episodeMatch.index)
                : query
        ).replace(/[：:：\-—]+$/, '');

        return {
            title: title || query,
            episodeNumber: Number.isFinite(episodeNumber) ? episodeNumber : null,
            indexUnit
        };
    }

    function extractEpisode(...values) {
        const text = values.filter(Boolean).join(' ');
        const patterns = [
            {pattern: /第\s*(\d{1,4})\s*([集期话部])/i, unitIndex: 2},
            {pattern: /(?:更新至|全)\s*(\d{1,4})\s*集/i, unit: '集'},
            {pattern: /\bEP?\s*0*(\d{1,4})\b/i, unit: '集'},
            {pattern: /(?:第)?\s*(\d{1,4})\s*期/i, unit: '期'},
            {pattern: /(?:^|_)0*(\d{1,4})(?:_|$)/, unit: '集'}
        ];
        for (const item of patterns) {
            const match = text.match(item.pattern);
            if (match) {
                const unit = item.unit || match[item.unitIndex] || '集';
                return `第${Number(match[1])}${unit}`;
            }
        }
        return '';
    }

    function findSelectedEpisodeText() {
        const nodes = document.querySelectorAll(
            '[aria-current="true"], [class*="active" i], [class*="current" i], [class*="selected" i]'
        );
        let best = '';
        let bestScore = 0;
        for (const node of nodes) {
            if (node.closest?.(`#${APP_ID}`)) continue;
            const text = cleanText(node.getAttribute?.('title') || node.textContent);
            if (!text || text.length > 180 || !/第\s*\d{1,4}\s*[集期话]/.test(text)) continue;
            let score = 10;
            if (/20\d{2}[-/.年]\d{1,2}[-/.月]\d{1,2}/.test(text)) score += 30;
            if (/特别加更|加更|纯享|先导|预告|花絮|彩蛋/.test(text)) score += 20;
            if (text.length >= 8 && text.length <= 100) score += 8;
            if (score > bestScore) {
                best = text;
                bestScore = score;
            }
        }
        return best;
    }

    function extractAirDate(...values) {
        const text = values.filter(Boolean).join(' ');
        const separated = text.match(
            /(20\d{2})[-/.年](\d{1,2})[-/.月](\d{1,2})(?:日)?/
        );
        if (separated) {
            return [
                separated[1],
                separated[2].padStart(2, '0'),
                separated[3].padStart(2, '0')
            ].join('');
        }
        return text.match(/\b(20\d{6})\b/)?.[1] || '';
    }

    function extractEpisodeVariant(...values) {
        const text = values.filter(Boolean).join(' ');
        const variants = [
            '特别加更',
            '加更',
            '纯享版',
            '纯享',
            '先导片',
            '先导',
            '预告',
            '花絮',
            '彩蛋'
        ];
        return variants.find((variant) => text.includes(variant)) || '正片';
    }

    function detectMediaType(...values) {
        const text = values.filter(Boolean).join(' ');
        const url = location.href;
        if (/[?&]theme=movie(?:&|$)/i.test(url) || /电影|movie/i.test(text)) {
            return 'movie';
        }
        if (/综艺|真人秀|variety/i.test(text)) return 'variety';
        if (/动漫|动画|国创|anime/i.test(text)) return 'anime';
        if (/电视剧|短剧|连续剧|tv series/i.test(text)) return 'tv';
        return '';
    }

    function extractReleaseYear(...values) {
        const text = values.filter(Boolean).join(' ');
        const year = text.match(/\b((?:19|20)\d{2})\b/)?.[1];
        return year || '';
    }

    function getPlatformInfo() {
        return PLATFORM_RULES.find((item) => item.host.test(location.hostname)) || null;
    }

    function extractMediaInfo() {
        const platform = getPlatformInfo();
        const structuredTitle = getStructuredPlatformTitle(platform);
        const domTitle = firstText(platform?.titleSelectors);
        const platformDocumentTitle = extractPlatformDocumentTitle(platform);
        const visibleTitle = findVisibleTitleCandidate();
        const ogTitle = metaContent(
            'meta[property="og:title"]',
            'meta[name="title"]',
            'meta[itemprop="name"]'
        );
        const episodeText = firstText(platform?.episodeSelectors) || findSelectedEpisodeText();
        const rawTitle = chooseBestMediaTitle([
            structuredTitle,
            platformDocumentTitle,
            domTitle,
            ogTitle,
            visibleTitle,
            document.title
        ]);
        const normalizedTitle = normalizeDocumentTitle(rawTitle);
        const episode = extractEpisode(episodeText, rawTitle, document.title);
        const title = stripTitleDecorations(parseSearchIntent(normalizedTitle).title);
        const episodeIntent = parseSearchIntent(episode);
        const episodeContext = [episodeText, rawTitle, document.title].filter(Boolean).join(' ');
        const pageMeta = [
            metaContent(
                'meta[property="og:type"]',
                'meta[itemprop="genre"]',
                'meta[name="genre"]'
            )
        ].filter(Boolean).join(' ');
        const releaseMeta = metaContent(
            'meta[property="video:release_date"]',
            'meta[itemprop="copyrightYear"]'
        );
        const episodeVariant = episode ? extractEpisodeVariant(episodeContext) : '';
        const query = cleanText([
            title,
            episode,
            episodeVariant && episodeVariant !== '正片' ? episodeVariant : ''
        ].filter(Boolean).join(' '));

        return {
            platform: platform?.name || location.hostname,
            title,
            episode,
            episodeNumber: episodeIntent.episodeNumber,
            indexUnit: episodeIntent.indexUnit,
            airDate: extractAirDate(episodeContext),
            episodeVariant,
            mediaType: detectMediaType(rawTitle, document.title, pageMeta),
            releaseYear: extractReleaseYear(rawTitle, document.title, releaseMeta),
            query,
            sourceUrl: location.href,
            createdAt: Date.now()
        };
    }

    function isRouteConfigured(route) {
        return Boolean(cleanText(route?.urlTemplate));
    }

    function getSettings() {
        const stored = GM_getValue(SETTINGS_KEY, {});
        const firstEnabledIndex = AUTHORIZED_ROUTES.findIndex(
            (route) => isRouteConfigured(route)
        );
        const storedIndex = Number(stored?.defaultRouteIndex);
        const validStoredRoute = isRouteConfigured(AUTHORIZED_ROUTES[storedIndex]);

        return {
            defaultAction: stored?.defaultAction === 'route' ? 'route' : 'target',
            defaultRouteIndex: validStoredRoute ? storedIndex : firstEnabledIndex,
            fadeSeconds: Number.isFinite(Number(stored?.fadeSeconds))
                ? Math.min(300, Math.max(0, Number(stored.fadeSeconds)))
                : DEFAULT_FADE_SECONDS,
            edgeSnap: stored?.edgeSnap !== false,
            launcherPosition: stored?.launcherPosition
                && Number.isFinite(Number(stored.launcherPosition.x))
                && Number.isFinite(Number(stored.launcherPosition.y))
                ? {
                    x: Math.min(1, Math.max(0, Number(stored.launcherPosition.x))),
                    y: Math.min(1, Math.max(0, Number(stored.launcherPosition.y)))
                }
                : null
        };
    }

        GM_setValue(SETTINGS_KEY, settings);
    }

            });

    function escapeHtml(value) {
        return String(value || '').replace(/[&<>"']/g, (char) => ({
            '&': '&amp;',
            '<': '&lt;',
            '>': '&gt;',
            '"': '&quot;',
            "'": '&#39;'
        })[char]);
    }

    }

    function buildAuthorizedRouteUrl(route, mediaInfo, query) {
        if (!isRouteConfigured(route)) return '';

        const values = {
            url: mediaInfo.sourceUrl,
            title: mediaInfo.title || query,
            episode: mediaInfo.episode,
            query
        };
        const hasPlaceholder = /\{(?:url|title|episode|query)\}/.test(route.urlTemplate);
        let rendered;
        if (hasPlaceholder) {
            rendered = route.urlTemplate.replace(
                /\{(url|title|episode|query)\}/g,
                (_, key) => encodeURIComponent(values[key] || '')
            );
        } else if (/[=?&/]$/.test(route.urlTemplate.trim())) {
            rendered = `${route.urlTemplate.trim()}${encodeURIComponent(values.url || '')}`;
        } else {
            throw new Error('线路地址需包含占位符，或以 =、?、&、/ 结尾作为接口前缀');
        }
        const parsed = new URL(rendered);
        if (parsed.protocol !== 'https:') {
            throw new Error('授权接口必须使用 HTTPS');
        }
        return parsed.href;
    }

    function waitForPlayerContainer(selector, timeoutMs = 12000) {
        const existing = document.querySelector(selector);
        if (existing) return Promise.resolve(existing);

        return new Promise((resolve, reject) => {
            const observer = new MutationObserver(() => {
                const container = document.querySelector(selector);
                if (!container) return;
                clearTimeout(timeout);
                observer.disconnect();
                resolve(container);
            });
            const timeout = window.setTimeout(() => {
                observer.disconnect();
                reject(new Error('未找到当前网站的原播放窗口'));
            }, timeoutMs);
            observer.observe(document.documentElement, {childList: true, subtree: true});
        });
    }

    function hidePlayerOverlays(selector) {
        if (!selector) return;
        document.querySelectorAll(selector).forEach((node) => {
            node.style.setProperty('display', 'none', 'important');
            node.style.setProperty('opacity', '0', 'important');
            node.style.setProperty('pointer-events', 'none', 'important');
        });
    }

    async function showAuthorizedRouteInPlayer(routeUrl) {
        const target = PLAYER_TARGETS.find((item) => item.host.test(location.hostname));
        if (!target) throw new Error('当前网站尚未配置原窗口播放位置');

        const container = await waitForPlayerContainer(target.container);
        const rect = container.getBoundingClientRect();
        const computed = getComputedStyle(container);
        const fallbackHeight = Math.max(
            180,
            Math.min(
                Math.round((rect.width || window.innerWidth) * 9 / 16),
                Math.round(window.innerHeight * .72)
            )
        );
        const playerHeight = rect.height >= 120 ? rect.height : fallbackHeight;

        hidePlayerOverlays(target.cleanup);
        container.querySelectorAll('video, audio').forEach((media) => {
            try {
                media.pause();
                media.removeAttribute('autoplay');
            } catch (error) {
                // 原站播放器可能代理了媒体方法；清空容器仍会停止其显示。
            }
        });

        const wrapper = document.createElement('div');
        wrapper.id = ROUTE_PLAYER_ID;
        Object.assign(wrapper.style, {
            position: 'absolute',
            inset: '0',
            zIndex: '2147483646',
            width: '100%',
            height: '100%',
            minHeight: `${playerHeight}px`,
            overflow: 'hidden',
            background: '#000'
        });

        const iframe = document.createElement('iframe');
        iframe.src = routeUrl;
        iframe.title = '授权线路播放器';
        iframe.allow = 'autoplay; encrypted-media; fullscreen; picture-in-picture';
        iframe.allowFullscreen = true;
        Object.assign(iframe.style, {
            display: 'block',
            width: '100%',
            height: '100%',
            minHeight: `${playerHeight}px`,
            border: '0',
            background: '#000'
        });
        wrapper.appendChild(iframe);

        if (computed.position === 'static') container.style.position = 'relative';
        container.style.setProperty('overflow', 'hidden', 'important');
        container.style.setProperty('min-height', `${playerHeight}px`, 'important');
        container.replaceChildren(wrapper);
        container.scrollIntoView({block: 'center', behavior: 'smooth'});
    }

    function normalizeMatchText(value) {
        return cleanText(value)
            .toLocaleLowerCase()
            .replace(/[\s·:：\-—_《》【】()[\]（）]/g, '');
    }

    function getEpisodeDescriptorFromLink(link) {
        const labelParts = [
            link.querySelector?.('.episode-ctn')?.getAttribute('title'),
            link.getAttribute?.('title'),
            link.textContent
        ]
            .map((value) => cleanText(value))
            .filter(Boolean);
        const label = [...new Set(labelParts)].join(' ');
        const match = label.match(/第\s*0*(\d{1,4})\s*[集期话部]/)
            || labelParts
                .map((part) => part.match(/^0*(\d{1,4})$/))
                .find(Boolean);
        return {
            number: match ? Number(match[1]) : null,
            airDate: extractAirDate(label),
            variant: extractEpisodeVariant(label),
            label
        };
    }

    function findEpisodeLink(root, pending) {
        const episodeNumber = Number(pending?.episodeNumber);
        const wantedDate = pending?.airDate || '';
        const wantedVariant = pending?.episodeVariant || '正片';
        let best = null;

        [...root.querySelectorAll('a[href*="/play/"]')].forEach((link) => {
            const descriptor = getEpisodeDescriptorFromLink(link);
            if (descriptor.number !== episodeNumber) return;
            if (wantedVariant && descriptor.variant !== wantedVariant) return;

            let score = 100;
            if (wantedDate) {
                if (descriptor.airDate === wantedDate) score += 240;
                else if (descriptor.airDate) score -= 180;
                else score -= 35;
            }
            if (descriptor.variant === wantedVariant) {
                score += 120;
            }
            if (descriptor.variant === '正片') score += 20;

            if (!best || score > best.score) best = {link, score, descriptor};
        });
        return best?.link || null;
    }

    function getTargetResultInfo(link) {
        const container = link.closest(
            'app-video-teaser-searchresult, .video-detail'
        ) || link.parentElement;
        const text = cleanText(container?.textContent);
        const episodeCount = container
            ? container.querySelectorAll('a[href*="/play/"] .episode-ctn').length
            : 0;
        let mediaType = '';
        if (/电影/.test(text)) mediaType = 'movie';
        else if (/综艺|真人秀/.test(text)) mediaType = 'variety';
        else if (/动漫|动画|国创/.test(text)) mediaType = 'anime';
        else if (/电视剧|短剧|连续剧/.test(text)) mediaType = 'tv';
        return {
            container,
            text,
            episodeCount,
            mediaType,
            releaseYear: extractReleaseYear(text)
        };
    }

    function selectExactTargetTitleLink(pending) {
        const wantedTitle = normalizeMatchText(pending?.title);
        const candidates = [...document.querySelectorAll('a.result-nav-link')]
            .filter((link) => (
                normalizeMatchText(link.textContent) === wantedTitle
                && /\/play\//.test(link.href)
            ))
            .map((link) => {
                const info = getTargetResultInfo(link);
                let score = 100;
                if (pending.mediaType) {
                    if (info.mediaType === pending.mediaType) score += 260;
                    else if (info.mediaType) score -= 260;
                }
                if (pending.mediaType === 'movie') {
                    score += info.episodeCount > 1 ? -320 : 100;
                } else if (pending.mediaType && info.episodeCount > 0) {
                    score += 40;
                }
                if (pending.releaseYear) {
                    if (info.releaseYear === pending.releaseYear) score += 140;
                    else if (info.releaseYear) score -= 80;
                }
                return {link, info, score};
            })
            .sort((left, right) => right.score - left.score);

        if (!candidates.length) return null;
        const best = candidates[0];
        const incompatibleMovie = pending.mediaType === 'movie'
            && (best.info.episodeCount > 1 || (best.info.mediaType && best.info.mediaType !== 'movie'));
        const incompatibleType = pending.mediaType
            && best.info.mediaType
            && best.info.mediaType !== pending.mediaType;

        return {
            ...best,
            compatible: !incompatibleMovie && !incompatibleType
        };
    }

    function findExactTargetDestination(pending) {
        const wantedTitle = normalizeMatchText(pending?.title);
        if (!wantedTitle) return null;

        const selectedResult = selectExactTargetTitleLink(pending);
        if (!selectedResult) return null;
        if (!selectedResult.compatible) {
            return {
                blocked: true,
                label: `找到同名《${pending.title}》，但内容类型与源页面不一致，已停止自动跳转`
            };
        }
        const {link: titleLink, info} = selectedResult;

        const episodeNumber = Number(pending?.episodeNumber);
        if (!Number.isFinite(episodeNumber) || episodeNumber <= 0) {
            return {href: titleLink.href, label: pending.title};
        }

        const resultContainer = info.container;
        if (!resultContainer) return null;

        const episodeLink = findEpisodeLink(resultContainer, pending);

        const unit = pending.indexUnit || '集';
        return episodeLink
            ? {href: episodeLink.href, label: `${pending.title} 第${episodeNumber}${unit}`}
            : {
                href: titleLink.href,
                label: `进入《${pending.title}》完整选集继续定位`,
                intermediate: true
            };
    }

    function autoOpenExactTarget(pending, root) {
        const isSearchPage = location.pathname.startsWith('/search/');
        const isSeriesPage = location.pathname.startsWith('/play/')
            && pending?.stage === 'series';
        if (
            !pending
            || pending.completedAt
            || (!isSearchPage && !isSeriesPage)
            || Date.now() - Number(pending.createdAt || 0) > 10 * 60 * 1000
        ) {
            return;
        }

        let attempts = 0;
        const timer = setInterval(() => {
            attempts += 1;
            const destination = isSearchPage
                ? findExactTargetDestination(pending)
                : (() => {
                    const episodeNumber = Number(pending.episodeNumber);
                    const episodeLink = findEpisodeLink(document, pending);
                    return episodeLink
                        ? {
                            href: episodeLink.href,
                            label: `${pending.title} 第${episodeNumber}${pending.indexUnit || '集'}`
                        }
                        : null;
                })();
            if (destination?.blocked) {
                clearInterval(timer);
                setStatus(root, destination.label);
                root.classList.add('tc-open');
                return;
            }
            if (destination?.href) {
                clearInterval(timer);
                if (destination.intermediate) {
                    GM_setValue(PENDING_KEY, {
                        ...pending,
                        stage: 'series',
                        seriesUrl: destination.href
                    });
                    setStatus(root, destination.label);
                    location.assign(destination.href);
                    return;
                }
                GM_setValue(PENDING_KEY, {
                    ...pending,
                    stage: 'complete',
                    completedAt: Date.now()
                });
                setStatus(root, `已精确定位：${destination.label}`);
                location.assign(destination.href);
                return;
            }
            if (attempts >= 15) {
                clearInterval(timer);
                setStatus(
                    root,
                    isSeriesPage
                        ? `已进入《${pending.title}》，但完整选集尚未加载出第${pending.episodeNumber}${pending.indexUnit || '集'}`
                        : '未找到完全同名资源，已停留在搜索结果页供你确认'
                );
                root.classList.add('tc-open');
            }
        }, 400);
    }

            ? `来源：${pending.platform}　目标：${pending.title || query}${targetDetails ? ` · ${targetDetails}` : ''}`
            : '未收到来源页面信息，可手动输入关键词';

        let targetFieldsEdited = false;
        [input, indexNumber, indexUnit].forEach((field) => {
            field.addEventListener('input', () => {
                targetFieldsEdited = true;
            });
            field.addEventListener('change', () => {
                targetFieldsEdited = true;
            });
        });

        root.querySelector('[data-action="fill"]').addEventListener('click', () => {
            const title = stripTitleDecorations(
                cleanText(input.value)
                || pending?.title
                || parseSearchIntent(query).title
            );
            const typedNumber = Number.parseInt(indexNumber.value, 10);
            const episodeNumber = Number.isFinite(typedNumber) && typedNumber > 0
                ? typedNumber
                : pending?.episodeNumber;
            const selectedUnit = indexUnit.value || pending?.indexUnit || (episodeNumber ? '集' : '');
            if (!title) {
                setStatus(root, '请输入片名或节目名');
                return;
            }
            const changedTitle = Boolean(
                cleanText(input.value)
                && normalizeMatchText(input.value) !== normalizeMatchText(pending?.title)
            );
            const nextPending = {
                ...(pending || {}),
                title,
                episodeNumber: episodeNumber || null,
                indexUnit: selectedUnit,
                episodeVariant: targetFieldsEdited && episodeNumber
                    ? '正片'
                    : (pending?.episodeVariant || (episodeNumber ? '正片' : '')),
                airDate: targetFieldsEdited ? '' : (pending?.airDate || ''),
                mediaType: changedTitle ? '' : (pending?.mediaType || ''),
                releaseYear: changedTitle ? '' : (pending?.releaseYear || ''),
                query: cleanText([
                    title,
                    episodeNumber ? `第${episodeNumber}${selectedUnit || '集'}` : ''
                ].filter(Boolean).join(' ')),
                createdAt: Date.now(),
                completedAt: null,
                stage: null
            };
            GM_setValue(PENDING_KEY, nextPending);
            setStatus(root, `正在重新定位：${nextPending.query}`);
            location.assign(`${TARGET_HOME}search/${encodeURIComponent(title)}`);
        });

        if (query) root.classList.add('tc-open');
        autoOpenExactTarget(pending, root);
    }

    if (location.hostname === TARGET_HOST) renderTargetPage();
    else renderSourcePage();
})();
